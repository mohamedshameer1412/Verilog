Student Mark Sheet
Arrays · foreach · grade calculation
ARRAY
Objective
Store marks of five subjects using an array and calculate the total, average, and grade.
// Program 1: Student marks - total, average, grade
module student_marks;
  int    marks[5]; int total; real average; string grade;
  initial begin
    marks[0]=85; marks[1]=72; marks[2]=90; marks[3]=65; marks[4]=78;
    total=0; foreach(marks[i]) total+=marks[i];
    average=real'(total)/5.0;
    if(average>=90) grade="A+"; else if(average>=80) grade="A";
    else if(average>=70) grade="B"; else if(average>=60) grade="C";
    else if(average>=50) grade="D"; else grade="F";
    $display("===== Student Mark Sheet =====");
    foreach(marks[i]) $display("Subject %0d : %0d",i+1,marks[i]);
    $display("Total   : %0d / 500",total);
    $display("Average : %.2f",average);
    $display("Grade   : %s",grade);
  end
endmodule
module tb_student_marks; student_marks dut(); initial begin #10; $finish; end endmodule

Key Concepts

  • int marks[5] — fixed-size unpacked int array
  • foreach(marks[i]) — auto-iterates all elements
  • real'(total) — type cast for fractional division

Temperature Sensor Array
Real arrays · min/max/avg
ARRAY
Objective
Store temperature readings from multiple sensors and determine minimum, maximum, and average temperature.
module temperature_sensor;
  real temp[8]; real min_t,max_t,sum_t,avg_t;
  initial begin
    temp[0]=23.5;temp[1]=31.2;temp[2]=18.7;temp[3]=27.0;
    temp[4]=35.1;temp[5]=22.4;temp[6]=29.8;temp[7]=16.3;
    min_t=temp[0]; max_t=temp[0]; sum_t=0.0;
    foreach(temp[i]) begin
      sum_t+=temp[i];
      if(temp[i]<min_t) min_t=temp[i];
      if(temp[i]>max_t) max_t=temp[i];
    end
    avg_t=sum_t/8.0;
    $display("Sensor Report");
    foreach(temp[i]) $display("Sensor %0d : %.1f C",i,temp[i]);
    $display("Min: %.1f  Max: %.1f  Avg: %.2f",min_t,max_t,avg_t);
  end
endmodule

FIFO Buffer using Queue
push_back · pop_front · overflow guard
QUEUE
Objective
Implement a FIFO buffer using a queue and demonstrate insertion and deletion of data.
module fifo_queue;
  int fifo[$]; int item; int capacity=5;
  task automatic push(input int data);
    if(fifo.size()<capacity) begin fifo.push_back(data);
      $display("[PUSH] %0d | size=%0d | %p",data,fifo.size(),fifo);
    end else $display("[PUSH FAIL] FIFO Full! %0d",data);
  endtask
  task automatic pop();
    if(fifo.size()>0) begin item=fifo.pop_front();
      $display("[POP ] %0d | size=%0d | %p",item,fifo.size(),fifo);
    end else $display("[POP FAIL] FIFO Empty!");
  endtask
  initial begin
    push(10);push(20);push(30);push(40);push(50);push(60);
    pop();pop();pop();
    push(70);push(80);
    pop();pop();pop();pop();pop();
  end
endmodule

ALU with Case Statement
ADD · SUB · MUL · DIV · MOD
CASE
Objective
Accept two operands and an operation code. Use a case statement to perform addition, subtraction, multiplication, division, and modulus.
module alu_case;
  int a,b,result; logic[2:0] op;
  initial begin
    a=20; b=6;
    $display("Operands A=%0d B=%0d",a,b);
    for(int code=0;code<=4;code++) begin
      op=code;
      case(op)
        3'd0: begin result=a+b; $display("ADD: %0d",result); end
        3'd1: begin result=a-b; $display("SUB: %0d",result); end
        3'd2: begin result=a*b; $display("MUL: %0d",result); end
        3'd3: begin result=a/b; $display("DIV: %0d",result); end
        3'd4: begin result=a%b; $display("MOD: %0d",result); end
        default: $display("Unknown");
      endcase
    end
  end
endmodule

  • case(op) — multi-way decode of operation code
  • for loop drives op codes 0-4 sequentially
  • Integer % operator for modulus

Model a traffic signal using enum. Use procedural statements and timing to implement RED → GREEN → YELLOW → RED.
module traffic_signal;
  typedef enum logic[1:0]{RED=2'b00,GREEN=2'b01,YELLOW=2'b10} state_t;
  state_t cur; int cycle;
  initial begin

      cur=RED;    $display("%5t  %-8s STOP",$time,cur.name()); #30;

      cur=YELLOW; $display("%5t  %-8s CAUTION",$time,cur.name()); #5;

    $display("Sequence complete");
  end

Concurrent Processes — fork/join Variants
fork...join · fork...join_any · fork...join_none
FORK/JOIN
Objective
Three concurrent processes (generate, process, monitor). Demonstrate fork...join, fork...join_any, and fork...join_none.
module fork_join_demo;
  int shared_data;
  task automatic gen_data();  #5;  shared_data=$urandom_range(1,100);
    $display("[%0t] GEN : data=%0d",$time,shared_data); endtask
  task automatic proc_data(); #10;
    $display("[%0t] PROC: x2=%0d",$time,shared_data*2); endtask
  task automatic mon_data();  #15;
    $display("[%0t] MON : done",$time); endtask

  initial begin
    $display("=== fork...join (wait ALL) ===");
    fork gen_data(); proc_data(); mon_data(); join
    $display("[%0t] All threads done\n",$time);

    $display("=== fork...join_any (wait FIRST) ===");
    fork
      begin #3;  $display("[%0t] A done",$time); end
      begin #7;  $display("[%0t] B done",$time); end
      begin #12; $display("[%0t] C done",$time); end
    join_any
    $display("[%0t] join_any returned\n",$time);

    $display("=== fork...join_none (no wait) ===");
    fork
      begin #2; $display("[%0t] BG1 done",$time); end
      begin #4; $display("[%0t] BG2 done",$time); end
    join_none
    $display("[%0t] join_none returned immediately",$time);
    #10; $display("[%0t] BG tasks finished",$time);
  end
endmodule
module tb_fork; fork_join_demo dut(); initial begin #500; $finish; end endmodule

  • fork...join_any — blocks until FIRST thread finishes; others continue

  • task automatic — each fork invocation gets its own stack frame

16x8 Memory with Read/Write Tasks
Tasks with output · boundary checking
TASKS
Objective
Model a 16x8-bit memory, create separate tasks for writing/reading. Perform multiple read/write operations and display results.
module memory_task;
  logic[7:0] mem[0:15];

  task automatic mem_write(input int addr, input logic[7:0] data);
    if(addr>=0 && addr<=15) begin
      mem[addr]=data; $display("[WR] Addr[%2d] <- 0x%02h",addr,data);
    end else $display("[WR] ERROR: addr %0d out of range!",addr);
  endtask

  task automatic mem_read(input int addr, output logic[7:0] data);
    if(addr>=0 && addr<=15) begin
      data=mem[addr]; $display("[RD] Addr[%2d] -> 0x%02h",addr,data);
    end else begin data=8'hXX; $display("[RD] ERROR: addr %0d out of range!",addr); end
  endtask

  logic[7:0] rdata;
  initial begin
    for(int i=0;i<16;i++) mem[i]=8'h00;
    mem_write(0,8'hAA); mem_write(3,8'h55); mem_write(7,8'hFF);
    mem_write(12,8'h1F); mem_write(15,8'hDE);
    mem_read(0,rdata); mem_read(3,rdata); mem_read(7,rdata);
    mem_read(12,rdata); mem_read(15,rdata);
    mem_write(7,8'h42); mem_read(7,rdata);
    mem_write(16,8'hBB); mem_read(-1,rdata);
  end
endmodule

4-bit ALU — Function + Task
ADD · SUB · AND · OR · XOR + PASS/FAIL verification
FUNCTION+TASK
Objective
Develop a 4-bit ALU supporting ADD, SUB, AND, OR, XOR. Use a function to compute result and a task to display and verify.
module alu_4bit;
  typedef enum logic[2:0]{ADD=0,SUB=1,AND_OP=2,OR_OP=3,XOR_OP=4} alu_op_t;

  function automatic logic[4:0] alu_compute(input logic[3:0] a,b, input alu_op_t op);
    logic[4:0] res;
    case(op)
      ADD:    res={1'b0,a}+{1'b0,b};
      SUB:    res={1'b0,a}-{1'b0,b};
      AND_OP: res={1'b0,a&b};
      OR_OP:  res={1'b0,a|b};
      XOR_OP: res={1'b0,a^b};
      default: res=5'bx;
    endcase
    return res;
  endfunction

  task automatic display_result(input logic[3:0] a,b, input alu_op_t op, input logic[4:0] res);
    string op_str; logic[4:0] expected;
    case(op)
      ADD:    begin op_str="ADD"; expected={1'b0,a}+{1'b0,b}; end
      SUB:    begin op_str="SUB"; expected={1'b0,a}-{1'b0,b}; end
      AND_OP: begin op_str="AND"; expected={1'b0,a&b}; end
      OR_OP:  begin op_str="OR";  expected={1'b0,a|b}; end
      XOR_OP: begin op_str="XOR"; expected={1'b0,a^b}; end
    endcase
    $display("%-3s A=%04b(%2d) B=%04b(%2d) Res=%05b(%2d) %s",
             op_str,a,a,b,b,res,res,(res===expected)?"PASS":"FAIL");
  endtask

  logic[3:0] a,b; logic[4:0] result;
  initial begin
    a=4'd9; b=4'd5; $display("Test A=%0d B=%0d",a,b);
    for(int i=0;i<5;i++) begin alu_op_t op=alu_op_t'(i);
      result=alu_compute(a,b,op); display_result(a,b,op,result); end
    a=4'd15; b=4'd3; $display("Test A=%0d B=%0d",a,b);
    for(int i=0;i<5;i++) begin alu_op_t op=alu_op_t'(i);
      result=alu_compute(a,b,op); display_result(a,b,op,result); end
  end
endmodule
